<script setup>
import { ref, reactive, computed, onMounted } from 'vue'
import { message } from 'ant-design-vue'

const configList = ref([])
const selectedConfig = ref('')
const loading = ref(false)
const listLoading = ref(false)
const saving = ref(false)
const activeTab = ref('form')
const rawJsonText = ref('')
const jsonError = ref('')

const periodOptions = [
  { label: '每日', value: '每日' },
  { label: '周一', value: '周一' },
  { label: '周二', value: '周二' },
  { label: '周三', value: '周三' },
  { label: '周四', value: '周四' },
  { label: '周五', value: '周五' },
  { label: '周六', value: '周六' },
  { label: '周日', value: '周日' }
]

const completionOptions = [
  { label: '�?, value: '�? },
  { label: '关闭游戏和软�?, value: '关闭游戏和软�? },
  { label: '关机', value: '关机' }
]

const resinKeys = ['浓缩树脂', '原粹树脂', '须臾树脂', '脆弱树脂']

const weekdayFields = [
  { label: '周一', partyKey: 'MondayPartyName', domainKey: 'MondayDomainName' },
  { label: '周二', partyKey: 'TuesdayPartyName', domainKey: 'TuesdayDomainName' },
  { label: '周三', partyKey: 'WednesdayPartyName', domainKey: 'WednesdayDomainName' },
  { label: '周四', partyKey: 'ThursdayPartyName', domainKey: 'ThursdayDomainName' },
  { label: '周五', partyKey: 'FridayPartyName', domainKey: 'FridayDomainName' },
  { label: '周六', partyKey: 'SaturdayPartyName', domainKey: 'SaturdayDomainName' },
  { label: '周日', partyKey: 'SundayPartyName', domainKey: 'SundayDomainName' }
]

const taskColumns = [
  { title: '编号', dataIndex: 'key', width: 90 },
  { title: '任务内容', dataIndex: 'label' },
  { title: '启用', dataIndex: 'enabled', width: 120 }
]

const createDefaultPeriodList = () => ({
  每日: false,
  周一: false,
  周二: false,
  周三: false,
  周四: false,
  周五: false,
  周六: false,
  周日: false
})

const createDefaultResinCount = () => ({
  浓缩树脂: 0,
  原粹树脂: 0,
  须臾树脂: 0,
  脆弱树脂: 0
})

const createDefaultConfig = () => ({
  EverySelectedValueListConverter: '',
  TaskEnabledList: {},
  GenshinUid: '',
  AccountBinding: false,
  Version: 1,
  Name: '',
  IndexId: 0,
  NextConfiguration: false,
  NextTaskIndex: 0,
  Period: '每日',
  PeriodList: createDefaultPeriodList(),
  ScheduleName: '',
  CustomDomainList: [],
  CraftingBenchCountry: '',
  AdventurersGuildCountry: '',
  PartyName: '',
  DomainName: '',
  WeeklyDomainEnabled: false,
  DailyRewardPartyName: '',
  MinResinToKeep: 0,
  SundayEverySelectedValue: '',
  SundaySelectedValue: '',
  SereniteaPotTpType: '',
  SecretTreasureObjects: [],
  ResinCount: createDefaultResinCount(),
  SpecifyResinUse: false,
  AccountBindingCode: '',
  MondayPartyName: '',
  MondayDomainName: '',
  TuesdayPartyName: '',
  TuesdayDomainName: '',
  WednesdayPartyName: '',
  WednesdayDomainName: '',
  ThursdayPartyName: '',
  ThursdayDomainName: '',
  FridayPartyName: '',
  FridayDomainName: '',
  SaturdayPartyName: '',
  SaturdayDomainName: '',
  SundayPartyName: '',
  SundayDomainName: '',
  CompletionAction: ''
})

const currentConfig = reactive(createDefaultConfig())

const periodCheckboxOptions = periodOptions.map(item => ({ label: item.label, value: item.value }))

const selectedPeriods = computed({
  get() {
    const list = currentConfig.PeriodList || {}
    return Object.keys(list).filter(key => list[key])
  },
  set(values) {
    if (!currentConfig.PeriodList) {
      currentConfig.PeriodList = createDefaultPeriodList()
    }
    Object.keys(currentConfig.PeriodList).forEach(key => {
      currentConfig.PeriodList[key] = values.includes(key)
    })
    values.forEach(day => {
      if (!(day in currentConfig.PeriodList)) {
        currentConfig.PeriodList[day] = true
      }
    })
  }
})

const numberParser = value => parseInt(value.replace(/[^\d]/g, ''), 10) || 0

const configOptions = computed(() =>
  configList.value.map(name => ({
    label: name,
    value: name
  }))
)

const filterConfigOption = (input, option) => {
  if (!option?.value) return false
  return option.value.toLowerCase().includes(input.toLowerCase())
}

const getTaskEnabled = taskKey => {
  const task = currentConfig.TaskEnabledList?.[taskKey]
  if (!task || typeof task !== 'object') return false
  if (typeof task.Item1 === 'boolean') return task.Item1
  if (typeof task.item1 === 'boolean') return task.item1
  return false
}

const setTaskEnabled = (taskKey, value) => {
  if (!currentConfig.TaskEnabledList || typeof currentConfig.TaskEnabledList !== 'object') {
    currentConfig.TaskEnabledList = {}
  }
  const task = currentConfig.TaskEnabledList[taskKey]
  if (!task || typeof task !== 'object') {
    currentConfig.TaskEnabledList[taskKey] = { Item1: value, Item2: '' }
    return
  }
  if ('Item1' in task) {
    task.Item1 = value
  } else if ('item1' in task) {
    task.item1 = value
  } else {
    task.Item1 = value
  }
}

const taskRows = computed(() => {
  const list = currentConfig.TaskEnabledList
  if (!list || typeof list !== 'object') return []
  return Object.entries(list)
    .map(([key, value]) => ({
      key,
      label: value?.Item2 || value?.item2 || key,
      enabled: getTaskEnabled(key)
    }))
    .sort((a, b) => {
      const numA = Number(a.key)
      const numB = Number(b.key)
      if (!Number.isNaN(numA) && !Number.isNaN(numB)) {
        return numA - numB
      }
      return a.key.localeCompare(b.key, 'zh-CN')
    })
})

const taskStats = computed(() => {
  const rows = taskRows.value
  const enabled = rows.filter(item => item.enabled).length
  return {
    total: rows.length,
    enabled
  }
})

const sanitizeConfig = config => JSON.parse(JSON.stringify(config))

const setConfigData = data => {
  const defaults = createDefaultConfig()
  const normalized = {
    ...defaults,
    ...data,
    Name: data?.Name || selectedConfig.value || defaults.Name,
    PeriodList: { ...defaults.PeriodList, ...(data?.PeriodList || {}) },
    ResinCount: { ...defaults.ResinCount, ...(data?.ResinCount || {}) },
    TaskEnabledList:
      data?.TaskEnabledList && typeof data.TaskEnabledList === 'object'
        ? data.TaskEnabledList
        : {},
    CustomDomainList: Array.isArray(data?.CustomDomainList) ? data.CustomDomainList : [],
    SecretTreasureObjects: Array.isArray(data?.SecretTreasureObjects)
      ? data.SecretTreasureObjects
      : []
  }

  Object.keys(defaults).forEach(key => {
    currentConfig[key] = normalized[key]
  })

  rawJsonText.value = JSON.stringify(normalized, null, 2)
}

const apiRequest = async (url, options = {}) => {
  const response = await fetch(url, {
    headers: {
      'Content-Type': 'application/json',
      ...(options.headers || {})
    },
    ...options
  })

  if (!response.ok) {
    const text = await response.text()
    throw new Error(text || `请求失败，状态码 ${response.status}`)
  }

  return response.json()
}

const fetchConfigList = async () => {
  listLoading.value = true
  try {
    const res = await apiRequest('/api/oneLong/oneLongAllName')
    configList.value = res?.data || []
    if (!selectedConfig.value && configList.value.length > 0) {
      selectedConfig.value = configList.value[0]
      await loadConfig()
    }
  } catch (error) {
    message.error(`加载配置列表失败�?{error.message}`)
  } finally {
    listLoading.value = false
  }
}

const loadConfig = async () => {
  if (!selectedConfig.value) {
    message.warning('请先选择一个配�?)
    return
  }
  loading.value = true
  jsonError.value = ''
  try {
    const res = await apiRequest(`/api/oneLong/config?name=${encodeURIComponent(selectedConfig.value)}`)
    const parsed = typeof res === 'string' ? JSON.parse(res) : res
    setConfigData(parsed)
    message.success('配置加载成功')
  } catch (error) {
    jsonError.value = `解析失败�?{error.message}`
    message.error('加载配置失败')
  } finally {
    loading.value = false
  }
}

const saveConfig = async () => {
  if (!currentConfig.Name) {
    message.warning('请填�?Name 字段后再保存')
    return
  }

  saving.value = true
  try {
    const payload = sanitizeConfig(currentConfig)
    const res = await apiRequest('/api/oneLong/saveConfig', {
      method: 'POST',
      body: JSON.stringify(payload)
    })

    if (res?.status === 'success') {
      message.success('保存成功')
    } else {
      message.error(res?.message || '保存失败')
    }
  } catch (error) {
    message.error(`保存失败�?{error.message}`)
  } finally {
    saving.value = false
  }
}

const syncJsonFromForm = () => {
  rawJsonText.value = JSON.stringify(sanitizeConfig(currentConfig), null, 2)
  message.success('已同步最新表单到 JSON')
}

const applyJsonToForm = () => {
  if (!rawJsonText.value) {
    message.warning('JSON 内容为空')
    return
  }
  try {
    const parsed = JSON.parse(rawJsonText.value)
    setConfigData(parsed)
    message.success('JSON 已应用到表单')
    jsonError.value = ''
  } catch (error) {
    jsonError.value = `JSON 解析失败�?{error.message}`
    message.error('JSON 解析失败')
  }
}

const formatJsonText = () => {
  if (!rawJsonText.value) return
  try {
    rawJsonText.value = JSON.stringify(JSON.parse(rawJsonText.value), null, 2)
    message.success('已格式化 JSON')
  } catch (error) {
    jsonError.value = `JSON 解析失败�?{error.message}`
    message.error('JSON 解析失败')
  }
}

const minifyJsonText = () => {
  if (!rawJsonText.value) return
  try {
    rawJsonText.value = JSON.stringify(JSON.parse(rawJsonText.value))
    message.success('已压�?JSON')
  } catch (error) {
    jsonError.value = `JSON 解析失败�?{error.message}`
    message.error('JSON 解析失败')
  }
}

const handleConfigChange = value => {
  if (value) {
    loadConfig()
  } else {
    setConfigData(createDefaultConfig())
    rawJsonText.value = ''
  }
}

onMounted(() => {
  fetchConfigList()
})
</script>
